  <script>
    const message = document.getElementById("message");

    let x = 10;
    let y = 20;
    let vx = 3;
    let vy = 2;

    const gravity = 0.15;
    const bounce = 0.9;

    function animate() {
      const maxX = Math.max(
        0, window.innerWidth - message.offsetWidth
      );

      const maxY = Math.max(
        0, window.innerHeight - message.offsetHeight
      );

      vy += gravity;
      x += vx;
      y += vy;

      if (x >= maxX) {
        x = maxX;
        vx = -Math.abs(vx) * bounce;
      } else if (x <= 0) {
        x = 0;
        vx = Math.abs(vx) * bounce;
      }

      if (y >= maxY) {
        y = maxY;
        vy = -Math.abs(vy) * bounce;
      } else if (y <= 0) {
        y = 0;
        vy = Math.abs(vy) * bounce;
      }

      message.style.transform =
        `translate(${x}px, ${y}px)`;

      requestAnimationFrame(animate);
    }

    animate();
  </script>
